# Sudoku Solver Rules
# Copy these rules into your ruleset to solve Sudoku puzzles
# Use with the data model: <urn:square/N> ex:hasIndexI i; ex:hasIndexJ j; ex:hasValue v

# ==============================================================================
# Rule 1: Row Constraint
# A value can go in a cell if no other cell in the same row has that value
# ==============================================================================

PREFIX ex: <http://example.org/sudoku#>

RULE {
  ?cell ex:hasValue ?value .
}
WHERE {
  # Cell without a value
  ?cell ex:hasIndexI ?row ;
        ex:hasIndexJ ?col .
  FILTER NOT EXISTS { ?cell ex:hasValue ?anyValue }

  # Check which values are already used in this row
  FILTER NOT EXISTS {
    ?otherCell ex:hasIndexI ?row ;
               ex:hasValue ?value .
  }

  # Value must be 1-9
  VALUES ?value { 1 2 3 4 5 6 7 8 9 }
}

# ==============================================================================
# Rule 3: Box Constraint
# A value can go in a cell if no other cell in the same 3x3 box has that value
# ==============================================================================

PREFIX ex: <http://example.org/sudoku#>

RULE {
  ?cell ex:hasValue ?value .
}
WHERE {
  # Cell without a value
  ?cell ex:hasIndexI ?row ;
        ex:hasIndexJ ?col .
  FILTER NOT EXISTS { ?cell ex:hasValue ?anyValue }

  # Calculate box indices (0-2 for each dimension)
  BIND(FLOOR((?row - 1) / 3) AS ?boxRow)
  BIND(FLOOR((?col - 1) / 3) AS ?boxCol)

  # Check which values are already used in this box
  FILTER NOT EXISTS {
    ?otherCell ex:hasIndexI ?otherRow ;
               ex:hasIndexJ ?otherCol ;
               ex:hasValue ?value .
    FILTER(FLOOR((?otherRow - 1) / 3) = ?boxRow)
    FILTER(FLOOR((?otherCol - 1) / 3) = ?boxCol)
  }

  # Value must be 1-9
  VALUES ?value { 1 2 3 4 5 6 7 8 9 }
}


# ==============================================================================
# Rule 4: Naked Single (Combined Constraints)
# If a cell has only one possible value considering all constraints, assign it
# This is the most powerful rule and should solve most easy/medium puzzles
# ==============================================================================

PREFIX ex: <http://example.org/sudoku#>

RULE {
  ?cell ex:hasValue ?value .
}
WHERE {
  # Cell without a value
  ?cell ex:hasIndexI ?row ;
        ex:hasIndexJ ?col .
  FILTER NOT EXISTS { ?cell ex:hasValue ?anyValue }

  # Calculate box indices
  BIND(FLOOR((?row - 1) / 3) AS ?boxRow)
  BIND(FLOOR((?col - 1) / 3) AS ?boxCol)

  # Try each possible value
  VALUES ?value { 1 2 3 4 5 6 7 8 9 }

  # Value not used in row
  FILTER NOT EXISTS {
    ?otherCell ex:hasIndexI ?row ;
               ex:hasValue ?value .
  }

  # Value not used in column
  FILTER NOT EXISTS {
    ?otherCell ex:hasIndexJ ?col ;
               ex:hasValue ?value .
  }

  # Value not used in box

    ?otherCell ex:hasIndexI ?otherRow ;
               ex:hasIndexJ ?otherCol ;
